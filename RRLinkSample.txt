		var timeElement = document.createElement('span');
		timeElement.setAttribute('class', 'time');
		timeElement.appendChild(document.createTextNode(FormatNumber(date.getHours(), 2) + ':' + FormatNumber(date.getMinutes(), 2) + ':' + FormatNumber(date.getSeconds(), 2) + '.' + FormatNumber(date.getMilliseconds(), 3)));

		var spacerElement = document.createElement('span');
		spacerElement.setAttribute('class', 'spacer');
		spacerElement.appendChild(document.createTextNode(' '));

		var messageElement = document.createElement('span');
		messageElement.setAttribute('class', 'message');

		message = message + '';
		if (e) {
			message = message + e;
			if (e.stack != undefined) {
				message = message + '\n' + e.stack;
			} else if (e.fileName !== undefined && e.lineNumber !== undefined) {
				message = message + '\n' + e.fileName + ' (' + e.lineNumber + ')';
			}
		}

		var lines = message.replace(/([\s]*[\r]?\n)+$/, '').split(/[\r]?\n/);
		for (var i = 0; i < lines.length; i++) {
			messageElement.appendChild(document.createTextNode(lines[i]));
			messageElement.appendChild(document.createElement('br'));
		}

		var logElement = document.createElement('div');
		logElement.setAttribute('class', 'log log-' + type.toLowerCase());
		logElement.appendChild(iconElement);
		logElement.appendChild(timeElement);
		logElement.appendChild(spacerElement);
		logElement.appendChild(messageElement);

		output.appendChild(logElement);
		output.scrollTop = output.scrollHeight;
	};

		Create: function() {
			/// <summary>
			/// RealTimeRecorderLinkClient を作成します。
			/// </summary>
			/// <returns type="Boolean">作成が完了した場合は <c>true</c>。それ以外の場合は <c>false</c>。</returns>

			TraceLog(MessageType.Info, 'Create が呼び出されました。');
			try {
				// すでに作成済みの場合は削除
				if (linkClient) {
					// 正しく作成されていた場合のみ破棄するメッセージを表示
					if (IsPreparedLinkClient()) {
						TraceLog(MessageType.Warn, '作成済みの RealTimeRecorderLinkClient を破棄します。');
					}

					document.body.removeChild(linkClient);
					linkClient = null;
					channelController = null;
					errorHandlerRegistered = false;
				}

				TraceLog(MessageType.Info, 'RealTimeRecorderLinkClient を作成します。');

				// IE 用に RealTimeRecorderLinkClient のインスタンスを作成
				var client = document.createElement('object');
				client.style.width  = 0;
				client.style.height = 0;
				client.setAttribute('classid', 'CLSID:' + RRLinkClientClassId);
				document.body.appendChild(client);

				// IE でロードできた場合は Status プロパティのタイプが 'unknown' になる
				if (typeof(client.Status) !== 'undefined') {
					linkClient = client;
					TraceLog(MessageType.Info, 'IE 用のコードで RealTimeRecorderLinkClient を作成しました。');
					return true;
				}

				// 作成できていないのでオブジェクトを削除
				document.body.removeChild(client);

				// ActiveX hosting plugin for Firefox が使用できない場合
				if (!navigator.mimeTypes[ActiveXHostingPluginMimeType]) {
					TraceLog(MessageType.Error, 'IE 用のコードで RealTimeRecorderLinkClient が作成できませんでした。また、ActiveX hosting plugin for Firefox がインストールされていないか無効になっています。\nIE 以外のブラウザでは ActiveX hosting plugin for Firefox (http://code.google.com/p/ff-activex-host/) をインストールする必要があります。');
					return;
				}

				// イベントハンドラを準備
				for (var name in Handlers) {
					window['RRLinkClient_' + name] = Handlers[name];
				}

				// ActiveX hosting plugin for Firefox (Chorme も可) を使用してインスタンスを作成
				TraceLog(MessageType.Info, 'IE 用のコードで RealTimeRecorderLinkClient が作成できなかったため ActiveX hosting plugin for Firefox を使用して作成します。');
				client = document.createElement('object');
				client.setAttribute('class' , 'object object-firefox');
				client.setAttribute('type'  , ActiveXHostingPluginMimeType);
				client.setAttribute('clsid' , '{' + RRLinkClientClassId + '}');
				client.setAttribute('width' , 0);
				client.setAttribute('height', 0);
				// MEMO: Chrome の場合 logger が指定してあると終了時にクラッシュする
				client.setAttribute('logger'    , 'RRLinkClient_AxHostLogger');
				client.setAttribute('debugLevel', '100');
				client.setAttribute('event_OnError'        , 'RRLinkClient_OnError');
				client.setAttribute('event_OnStatusChanged', 'RRLinkClient_OnStatusChanged');
				document.body.appendChild(client);

				if (typeof(client.Status) === 'undefined') {
					// ユーザが操作でプラグインの実行を許可出来るようにオブジェクトを画面に表示したままにしておく
					linkClient = client;
					TraceLog(MessageType.Error, 'ActiveX hosting plugin for Firefox を使用して RealTimeRecorderLinkClient を作成しましたが、プラグインの実行が許可されていません。\nActiveX hosting plugin for Firefox の実行を許可してください。');
					return false;
				}

				linkClient = client;

				// ActiveX hosting plugin for Firefox ではイベントハンドラが登録済み
				errorHandlerRegistered = true;
				TraceLog(MessageType.Info, 'ActiveX hosting plugin for Firefox を使用して RealTimeRecorderLinkClient を作成しました。');
				return true;
			} catch (e) {
				TraceLog(MessageType.Error, 'Create でエラーが発生しました。', e);
				return false;

			}

				document.getElementById('RRLinkClientPauseRecordingButton').setAttribute('disabled', 'disabled');
				document.getElementById('RRLinkClientResumeRecordingButton').setAttribute('disabled', 'disabled');
				return true;
			} catch (e) {
				TraceLog(MessageType.Error, 'StopRecording でエラーが発生しました。', e);
				return false;
			} finally {
				TraceLog(MessageType.Info, 'StopRecording が終了しました。');
			}
		},


		LaunchChannelSwitcher: function(silent) {
			/// <summary>
			/// チャンネル振り分け・音声再生ツール（ChannelSwitcher.exe）を別プロセスとして起動します。
			/// silent が true の場合（起動時の自動実行など）は、失敗しても画面には表示しません。
			/// ボタンから明示的に呼び出した場合（silent が偽）は、失敗理由を警告として表示します。
			/// </summary>
			/// <param name="silent" type="Boolean" mayBeNull="true">失敗を画面に表示しない場合は <c>true</c>。</param>
			/// <returns type="Boolean">起動できた場合は <c>true</c>。それ以外の場合は <c>false</c>。</returns>

			var failureType = silent ? MessageType.Trace : MessageType.Warn;

			TraceLog(MessageType.Info, 'LaunchChannelSwitcher が呼び出されました。');
			try {
				// パスはテキストボックスから取得する（画面表示時に設定ファイルの値が入っている）
				var pathValue = document.getElementById('RRLinkClientChannelSwitcherPath').value;
				if (!pathValue) {
					pathValue = ChannelSwitcherRelativePath;
				}

				// 前後の空白と、Explorer の「パスのコピー」で付く二重引用符を取り除く
				pathValue = pathValue.replace(/^\s+|\s+$/g, '');
				if (pathValue.charAt(0) === '"' && pathValue.charAt(pathValue.length - 1) === '"') {
					pathValue = pathValue.substring(1, pathValue.length - 1);
				}

				var fso = new ActiveXObject('Scripting.FileSystemObject');
				// ドライブレター（C:\...）や UNC（\\...）で始まる場合は絶対パスとしてそのまま使う。
				// それ以外は、この HTA と同じフォルダからの相対パスとして解決する
				var isAbsolute = /^[a-zA-Z]:[\\\/]/.test(pathValue) || /^\\\\/.test(pathValue);
				var exePath = isAbsolute ? pathValue : fso.BuildPath(ScriptDir, pathValue);
				if (!fso.FileExists(exePath)) {
					// silent なら仮想オーディオデバイス／ChannelSwitcher.exe が未配置な環境向けに
					// Trace として記録するだけで画面には表示しない。ボタン操作時は Warn として表示する
					TraceLog(failureType, 'ChannelSwitcher.exe が見つかりません: ' + exePath);
					return false;
				}

				var shell = new ActiveXObject('WScript.Shell');
				shell.Run('"' + exePath + '"', 1, false);
				TraceLog(MessageType.Info, 'ChannelSwitcher.exe を起動しました: ' + exePath);
				return true;
			} catch (e) {
				// 同上の理由で silent なら画面には表示しない
				TraceLog(failureType, 'ChannelSwitcher.exe の起動に失敗しました。', e);
				return false;
			} finally {
				TraceLog(MessageType.Info, 'LaunchChannelSwitcher が終了しました。');
			}
		}
	};
})();


// ---------- サービス名・ChannelSwitcher パスの保存・読み込み ----------

function LoadSettings() {
	var serviceName = DefaultServiceName;
	var channelSwitcherPath = ChannelSwitcherRelativePath;
	try {
		if (SettingsPath) {
			var fso = new ActiveXObject('Scripting.FileSystemObject');
			if (fso.FileExists(SettingsPath)) {
				var ts = fso.OpenTextFile(SettingsPath, 1);
				var content = ts.ReadAll();
				ts.Close();

				var lines = content.split(/\r?\n/);
				for (var i = 0; i < lines.length; i++) {
					var eq = lines[i].indexOf('=');
					if (eq < 0) continue;
					var key = lines[i].substring(0, eq);
					var val = lines[i].substring(eq + 1);
					if (key === 'serviceName' && val.length) {
						serviceName = val;
					} else if (key === 'channelSwitcherPath' && val.length) {
						channelSwitcherPath = val;
					}
				}
			}
		}
	} catch (e) {
		// 設定ファイルが読めない場合は既定値をそのまま使う
	}
	document.getElementById('RRLinkClientServiceName').value = serviceName;
	document.getElementById('RRLinkClientChannelSwitcherPath').value = channelSwitcherPath;
}

function SaveSettings() {
	try {
		if (!SettingsPath) return;
		var fso = new ActiveXObject('Scripting.FileSystemObject');
		var folder = fso.GetParentFolderName(SettingsPath);
		if (folder && !fso.FolderExists(folder)) {
			fso.CreateFolder(folder);
		}
		var ts = fso.CreateTextFile(SettingsPath, true);
		ts.WriteLine('serviceName=' + document.getElementById('RRLinkClientServiceName').value);
		ts.WriteLine('channelSwitcherPath=' + document.getElementById('RRLinkClientChannelSwitcherPath').value);
		ts.Close();
	} catch (e) {
		// 保存に失敗しても致命的ではないため、ここでは何もしない
	}
}


// ---------- 接続（Create〜SetAutoControl）----------

function Connect() {
	/// <summary>
	/// Create〜SetAutoControl をまとめて実行します。
	/// </summary>

	if (!RRLinkClient.Create()) {
		return;
	}

	if (!RRLinkClient.Initialize()) {
		return;
	}

	if (!RRLinkClient.WaitForConnection()) {
		return;
	}

	if (!RRLinkClient.GetChannelController()) {
		return;
	}

	if (!RRLinkClient.SetAutoControl()) {
		return;
	}

	// 全ての初期処理が成功したら操作パネルを有効化
	document.getElementById('RRLinkClientStartRecordingButton').removeAttribute('disabled');
}

function Reconnect() {
	/// <summary>
	/// サービス名を変更した場合などに、いったん切断してから接続をやり直します。
	/// </summary>

	SaveSettings();
	RRLinkClient.Terminate();
	RRLinkClient.Release();
	Connect();
}


// ---------- 通話属性をまとめて設定するパネル ----------

function CreateAttributeRow() {
	var row = document.createElement('div');
	row.className = 'attr-row';

	var keyCell = document.createElement('div');
	keyCell.className = 'attr-cell attr-cell-key';

	var select = document.createElement('select');
	select.className = 'select';

	var blank = document.createElement('option');
	blank.value = '';
	blank.text = '';
	select.appendChild(blank);

	for (var i = 0; i < ConversationAttributeKeys.length; i++) {
		var opt = document.createElement('option');
		opt.value = ConversationAttributeKeys[i][0];
		opt.text = ConversationAttributeKeys[i][1];
		select.appendChild(opt);
	}
	keyCell.appendChild(select);
	row.appendChild(keyCell);

	var valueCell = document.createElement('div');
	valueCell.className = 'attr-cell attr-cell-value';

	var input = document.createElement('input');
	input.type = 'text';
	input.className = 'textbox';
	input.maxLength = 256;
	valueCell.appendChild(input);
	row.appendChild(valueCell);

	var removeCell = document.createElement('div');
	removeCell.className = 'attr-cell attr-cell-remove';

	var removeButton = document.createElement('input');
	removeButton.type = 'button';
	removeButton.className = 'button';
	removeButton.value = '削除';
	removeButton.onclick = function() {
		row.parentNode.removeChild(row);
	};
	removeCell.appendChild(removeButton);
	row.appendChild(removeCell);

	return row;
}

function AddAttributeRow() {
	document.getElementById('RRLinkClientAttributesRows').appendChild(CreateAttributeRow());
}

function OpenAttributesPanel() {
	var container = document.getElementById('RRLinkClientAttributesRows');
	if (container.childNodes.length === 0) {
		for (var i = 0; i < 3; i++) {
			AddAttributeRow();
		}
	}
	document.getElementById('RRLinkClientAttributesOverlay').style.display = 'block';
}

function CloseAttributesPanel() {
	document.getElementById('RRLinkClientAttributesOverlay').style.display = 'none';
}

function ApplyAttributes() {
	var rows = document.getElementById('RRLinkClientAttributesRows').childNodes;
	var pairs = [];
	for (var i = 0; i < rows.length; i++) {
		var selects = rows[i].getElementsByTagName('select');
		var inputs = rows[i].getElementsByTagName('input');
		if (!selects.length || !inputs.length) continue;

		var key = selects[0].value;
		if (!key) continue;

		var value = inputs[0].value;
		pairs.push({ key: key, value: value.length ? value : null });
	}

	if (!pairs.length) {
		CloseAttributesPanel();
		return;
	}

	if (RRLinkClient.SetConversationAttributes(pairs)) {
		RRLinkClient.FlushConversationAttributes();
	}
	CloseAttributesPanel();
}
